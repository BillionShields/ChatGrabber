ChatGrabberv3

Drag the button below into your bookmarks bar. Click it on any supported conversation to download a self-contained .html export.

chatgpt.com copilot.microsoft.com aistudio.google.com claude.ai — new

Install

📥 ChatGrabber Drag onto your bookmarks bar — don't click.
No bar visible? Ctrl/Cmd + Shift + B.
Bookmarks bar not showing, or drag not working? Copy the code instead
  1. Create a new bookmark (any page, any name).
  2. Edit it and paste the code below as the URL.
  3. Save, then click that bookmark on a conversation page.

Use it

  1. Open a conversation on any supported site.
  2. Click the bookmarklet. A small status toast appears bottom-right.
  3. It auto-scrolls the conversation to force-load everything, then downloads <title>_ChatExport.html.

What's new in v3

Claude.ai support added. One caveat: I could only verify this against a saved, empty claude.ai/new page — no live conversation was available to confirm Claude's exact message markup, so the Claude.ai adapter unions several plausible selectors for the user/assistant message containers instead of betting on one, and dedupes if more than one guess matches the same message. Everything else (ChatGPT, Copilot, Gemini AI Studio) is unchanged from before.

If it downloads a file with only 1–2 messages, or the "no conversation found" alert fires on a real Claude.ai chat, that means the guessed selectors missed — send me the output and I'll tighten it against the real markup.

(async () => {
  const TOOL_NAME = "ChatGrabber";

  const sleep = (milliseconds) => new Promise((resolve) => setTimeout(resolve, milliseconds));
  const sanitizeFilename = (name) => name.replace(/[\\/:*?"<>|]+/g, "_").trim().slice(0, 100);
  const buildTimestamp = () => new Date().toISOString().replace(/[-:T]/g, "").slice(0, 15);

  // ---- status toast ----
  const statusToast = document.createElement("div");
  statusToast.title = TOOL_NAME;
  Object.assign(statusToast.style, {
    position: "fixed",
    bottom: "16px",
    right: "16px",
    zIndex: 2147483647,
    background: "#111",
    color: "#fff",
    font: "12px/1.4 -apple-system,Segoe UI,sans-serif",
    padding: "8px 12px",
    borderRadius: "8px",
    boxShadow: "0 2px 10px rgba(0,0,0,.35)",
    maxWidth: "260px",
    pointerEvents: "none",
  });
  const setStatus = (message) => {
    statusToast.textContent = `${TOOL_NAME}: ${message}`;
    if (!statusToast.isConnected) document.body.appendChild(statusToast);
  };

  // ---- site detection: ordered list of [siteName, matcher], first match wins ----
  // "copilot" stays last as the catch-all default (Copilot is served from several hostnames).
  const SITE_DETECTORS = [
    ["geminiai", () => location.hostname.includes("aistudio.google.com")],
    ["chatgpt", () => /chatgpt\.com|openai\.com/.test(location.hostname)],
    ["claudeai", () => location.hostname.includes("claude.ai")],
    ["copilot", () => true],
  ];
  const site = SITE_DETECTORS.find(([, matches]) => matches())[0];

  // ---- synthetic per-element IDs for sites that don't expose stable message IDs ----
  const syntheticIdsByElement = new WeakMap();
  const getOrCreateSyntheticId = (element) => {
    if (!syntheticIdsByElement.has(element)) {
      syntheticIdsByElement.set(element, `msg-${Math.random().toString(36).slice(2)}`);
    }
    return syntheticIdsByElement.get(element);
  };

  // ---- drop elements that are nested inside another already-matched element ----
  // Needed for claudeai, whose messageSelector unions several guessed selectors that
  // could theoretically overlap (e.g. a wrapper and an inner node both matching).
  const dedupeNestedElements = (elements) =>
    elements.filter((element, _, all) => !all.some((other) => other !== element && other.contains(element)));

  const SITE_CONFIGS = {
    chatgpt: {
      messageSelector: "[data-message-id][data-message-author-role]",
      getMessageId: (element) => element.getAttribute("data-message-id"),
      getRole: (element) => (element.getAttribute("data-message-author-role") === "user" ? "user" : "assistant"),
      getTitle: () => (document.querySelector("h1")?.textContent || document.title || "").trim(),
      scrollAnchorSelector: "[data-message-id][data-message-author-role], #thread",
    },
    copilot: {
      messageSelector: '[id$="-user-message"], [data-testid="ai-message"]',
      getMessageId: (element) =>
        element.id
          ? element.id
          : element.getAttribute("aria-labelledby")
          ? element.getAttribute("aria-labelledby")
          : getOrCreateSyntheticId(element),
      getRole: (element) => (element.matches('[data-testid="ai-message"]') ? "assistant" : "user"),
      getTitle: () => {
        const activeChat =
          document.querySelector("a[data-sidebar-item][data-active] span") ||
          document.querySelector("[aria-selected='true'] p[title]");
        return (activeChat ? activeChat.getAttribute("title") || activeChat.textContent : "").trim();
      },
      scrollAnchorSelector: '[id$="-user-message"], [data-testid="ai-message"], #thread, div[data-content="conversation"]',
    },
    // Claude.ai's exact DOM wasn't available to verify against a live conversation, so this
    // unions every plausible selector for each role rather than betting on one. Harmless if a
    // guess never matches; dedupeNestedElements() keeps a guessed wrapper/inner pair from
    // double-counting the same message if more than one guess happens to hit.
    claudeai: {
      messageSelector:
        '[data-testid="user-message"], .font-user-message, [data-testid="assistant-message"], .font-claude-message, .standard-markdown',
      getMessageId: (element) => getOrCreateSyntheticId(element),
      getRole: (element) => (element.matches('[data-testid="user-message"], .font-user-message') ? "user" : "assistant"),
      getTitle: () => {
        const fromDocumentTitle = (document.title || "").replace(/\s*[-–]\s*Claude\s*$/, "").trim();
        return fromDocumentTitle || (document.querySelector("h1")?.textContent || "").trim();
      },
      scrollAnchorSelector:
        '[data-testid="user-message"], .font-user-message, [data-testid="assistant-message"], .font-claude-message, .standard-markdown, main',
    },
    geminiai: {
      legacy: true,
      legacySelector: "ms-prompt-renderer",
      getTitle: () => (document.querySelector("h1.mode-title")?.textContent || "").trim(),
    },
  };
  const config = SITE_CONFIGS[site];

  // ---- merge two overlapping arrays of text lines (used to stitch scrolled CodeMirror captures) ----
  const mergeOverlappingLines = (previousLines, newLines) => {
    if (previousLines.length === 0) return newLines;
    for (let overlap = Math.min(previousLines.length, newLines.length); overlap > 0; overlap--) {
      if (previousLines.slice(-overlap).join("\n") === newLines.slice(0, overlap).join("\n")) {
        return previousLines.concat(newLines.slice(overlap));
      }
    }
    return previousLines.concat(newLines);
  };

  // ---- capture a CodeMirror 6 editor's full text by scrolling its internal viewport ----
  const captureCodeMirrorText = async (editorElement) => {
    const scroller = editorElement.querySelector(".cm-scroller") || editorElement;
    const readVisibleLines = () => Array.from(scroller.querySelectorAll(".cm-line")).map((line) => line.textContent);
    let capturedLines = readVisibleLines();
    if (scroller.scrollHeight > scroller.clientHeight + 4) {
      const originalScrollTop = scroller.scrollTop;
      scroller.scrollTop = 0;
      await sleep(220);
      capturedLines = readVisibleLines();
      let previousScrollTop = -1;
      let steps = 0;
      while (
        scroller.scrollTop < scroller.scrollHeight - scroller.clientHeight - 2 &&
        scroller.scrollTop !== previousScrollTop &&
        steps < 400
      ) {
        previousScrollTop = scroller.scrollTop;
        scroller.scrollTop = previousScrollTop + 0.8 * scroller.clientHeight;
        await sleep(220);
        capturedLines = mergeOverlappingLines(capturedLines, readVisibleLines());
        steps++;
      }
      scroller.scrollTop = originalScrollTop;
    }
    return capturedLines.join("\n");
  };

  const detectCodeLanguage = (preOrEditorElement) => {
    const languageLabel = preOrEditorElement.querySelector('[class*="justify-self-start"]');
    return (languageLabel ? languageLabel.textContent.trim() : "") || "text";
  };

  let codeMirrorIdCounter = 0;
  const codeMirrorCaptures = new Map();
  const captureCodeBlocksIn = async (messageElement) => {
    const editors = Array.from(messageElement.querySelectorAll(".cm-editor"));
    for (const editor of editors) {
      if (editor.dataset.grabberCmId) continue;
      const captureId = String(codeMirrorIdCounter++);
      editor.dataset.grabberCmId = captureId;
      const container = editor.closest("pre") || editor;
      try {
        const code = await captureCodeMirrorText(editor);
        codeMirrorCaptures.set(captureId, { language: detectCodeLanguage(container), code });
      } catch (error) {
        codeMirrorCaptures.set(captureId, { language: detectCodeLanguage(container), code: editor.textContent });
      }
    }
  };

  // ---- strip interactive/decorative chrome that shouldn't end up in the export ----
  const CHROME_SELECTOR = [
    "button",
    "svg",
    "textarea",
    "input",
    '[role="toolbar"]',
    '[aria-hidden="true"]',
    ".sr-only",
    '[data-testid*="reactions"]',
    '[data-testid$="-button"]',
    '[data-testid="sentinel-div"]',
    '[data-testid="backstage-chats"]',
  ].join(",");

  const ALLOWED_ATTRIBUTES_BY_TAG = {
    A: ["href", "title"],
    IMG: ["src", "alt"],
    TD: ["colspan", "rowspan"],
    TH: ["colspan", "rowspan"],
    CODE: ["class"],
  };
  const stripDisallowedAttributes = (root) => {
    root.querySelectorAll("*").forEach((element) => {
      const allowedAttributes = ALLOWED_ATTRIBUTES_BY_TAG[element.tagName] || [];
      Array.from(element.attributes).forEach((attribute) => {
        if (!allowedAttributes.includes(attribute.name)) element.removeAttribute(attribute.name);
      });
    });
  };

  // ---- build the final clean DOM node for one message: swap in the best-captured chunk
  // overrides (for sites that virtualize long replies) and CodeMirror captures, strip chrome ----
  const buildCleanMessageNode = (messageSkeleton, chunkOverridesById) => {
    const node = messageSkeleton.clone;
    node.querySelectorAll('[id*="-content-"]').forEach((chunkElement) => {
      const override = chunkOverridesById.get(chunkElement.id);
      if (override && override.textContent.trim().length > chunkElement.textContent.trim().length) {
        chunkElement.replaceWith(override.cloneNode(true));
      }
    });
    node.querySelectorAll(".cm-editor[data-grabber-cm-id]").forEach((editorElement) => {
      const capture = codeMirrorCaptures.get(editorElement.dataset.grabberCmId);
      const pre = document.createElement("pre");
      const code = document.createElement("code");
      code.className = capture ? `language-${capture.language.toLowerCase()}` : "";
      code.textContent = capture ? capture.code : editorElement.textContent;
      pre.appendChild(code);
      (editorElement.closest("pre") || editorElement).replaceWith(pre);
    });
    node.querySelectorAll(CHROME_SELECTOR).forEach((element) => element.remove());
    stripDisallowedAttributes(node);
    return node;
  };

  const blobToDataUrl = (blob) =>
    new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = () => resolve(reader.result);
      reader.onerror = reject;
      reader.readAsDataURL(blob);
    });

  const inlineImages = async (root) => {
    const dataUrlsBySrc = new Map();
    const images = Array.from(root.querySelectorAll("img[src]"));
    for (const image of images) {
      const absoluteSrc = new URL(image.getAttribute("src"), location.href).href;
      try {
        if (!dataUrlsBySrc.has(absoluteSrc)) {
          setStatus("inlining image…");
          const response = await fetch(absoluteSrc, { credentials: "include" });
          if (!response.ok) throw new Error(`HTTP ${response.status}`);
          const blob = await response.blob();
          if (blob.size > 8388608) throw new Error("image too large to inline");
          dataUrlsBySrc.set(absoluteSrc, await blobToDataUrl(blob));
        }
        image.src = dataUrlsBySrc.get(absoluteSrc);
      } catch (error) {
        image.src = absoluteSrc;
      }
    }
  };

  const escapeHtml = (text) => text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

  const buildExportHtml = (title, messages) => {
    const safeTitle = escapeHtml(title);
    const messagesHtml = messages
      .map(
        ({ role, html }) => `
      <div class="message ${role}">
        <div class="role-label">${role === "user" ? "You" : "Assistant"}</div>
        ${html}
      </div>`
      )
      .join("\n");
    return `<!DOCTYPE html>
<html>
<head>
<meta charset="utf-8">
<title>${safeTitle}</title>
<style>
    body { font: 15px/1.6 -apple-system,Segoe UI,Helvetica,Arial,sans-serif; max-width: 820px;
           margin: 32px auto; padding: 0 16px; color: #1a1a1a; background: #fff; }
    h1.export-title { font-size: 20px; margin-bottom: 4px; }
    .export-meta { color: #666; font-size: 12px; margin-bottom: 24px; }
    .message { border-radius: 12px; padding: 14px 18px; margin: 14px 0; }
    .message.user { background: #eef3ff; }
    .message.assistant { background: #f6f6f7; }
    .message .role-label { font-size: 11px; text-transform: uppercase; letter-spacing: .04em;
                            color: #888; margin-bottom: 6px; }
    .message img { max-width: 100%; border-radius: 8px; display: block; margin: 8px 0; }
    .message pre { background: #1e1e1e; color: #e6e6e6; padding: 12px 14px; border-radius: 8px;
                    overflow-x: auto; font: 13px/1.5 ui-monospace,Consolas,monospace; }
    .message code { font: 13px/1.5 ui-monospace,Consolas,monospace; }
    .message :not(pre) > code { background: #eee; color: #1a1a1a; padding: 1px 5px; border-radius: 4px; }
    .message table { border-collapse: collapse; width: 100%; margin: 10px 0; }
    .message th, .message td { border: 1px solid #ddd; padding: 6px 10px; text-align: left; }
    .message blockquote { border-left: 3px solid #ccc; margin: 8px 0; padding: 2px 12px; color: #555; }
  </style>
</head>
<body>
<h1 class="export-title">${safeTitle}</h1>
<div class="export-meta">Saved by ${TOOL_NAME} • ${new Date().toLocaleString()} • ${escapeHtml(location.href)}</div>
${messagesHtml}
</body>
</html>`;
  };

  const downloadHtml = (filename, html) => {
    const blob = new Blob([html], { type: "text/html" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.download = filename;
    link.title = TOOL_NAME;
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    URL.revokeObjectURL(url);
  };

  // ---- find the nearest scrollable ancestor of a message anchor element ----
  const findScrollableAncestor = (anchorElement) => {
    let current = anchorElement;
    while (current && current !== document.body && current !== document.documentElement) {
      const style = getComputedStyle(current);
      if (/(auto|scroll)/.test(style.overflowY) && current.scrollHeight > current.clientHeight + 4) return current;
      current = current.parentElement;
    }
    return document.scrollingElement || document.documentElement;
  };

  // ---- scroll the conversation top-to-bottom, capturing message skeletons + chunk overrides
  // at every step so virtualized/unmounted content gets picked up ----
  const scanConversation = async (scrollContainer) => {
    const messageSkeletonsById = new Map();
    const bestChunkById = new Map();

    const captureCurrentlyVisible = async () => {
      const visibleMessages = dedupeNestedElements(Array.from(document.querySelectorAll(config.messageSelector)));
      for (const messageElement of visibleMessages) {
        await captureCodeBlocksIn(messageElement);
        const messageId = config.getMessageId(messageElement);
        messageSkeletonsById.set(messageId, { role: config.getRole(messageElement), clone: messageElement.cloneNode(true) });
      }
      document.querySelectorAll('[id*="-content-"]').forEach((chunkElement) => {
        const chunkId = chunkElement.id;
        const chunkTextLength = chunkElement.textContent.trim().length;
        if (chunkTextLength === 0) return;
        const bestSoFar = bestChunkById.get(chunkId);
        if (!bestSoFar || chunkTextLength > bestSoFar.textContent.trim().length) {
          bestChunkById.set(chunkId, chunkElement.cloneNode(true));
        }
      });
    };

    const originalScrollTop = scrollContainer.scrollTop;
    scrollContainer.scrollTop = 0;
    await sleep(350);
    await captureCurrentlyVisible();

    let previousScrollTop = -1;
    let steps = 0;
    while (
      scrollContainer.scrollTop < scrollContainer.scrollHeight - scrollContainer.clientHeight - 2 &&
      scrollContainer.scrollTop !== previousScrollTop &&
      steps < 400
    ) {
      previousScrollTop = scrollContainer.scrollTop;
      setStatus(`scanning conversation… (${steps + 1})`);
      scrollContainer.scrollTop = previousScrollTop + 0.8 * scrollContainer.clientHeight;
      await sleep(220);
      await captureCurrentlyVisible();
      steps++;
    }

    await sleep(350);
    await captureCurrentlyVisible();
    scrollContainer.scrollTop = originalScrollTop;

    return { messageSkeletonsById, bestChunkById };
  };

  try {
    setStatus("starting…");

    if (config.legacy) {
      const legacyRoot = document.querySelector(config.legacySelector);
      if (!legacyRoot) return void alert(`${TOOL_NAME}: no conversation found on this page.`);
      const clone = legacyRoot.cloneNode(true);
      clone.querySelectorAll(CHROME_SELECTOR).forEach((element) => element.remove());
      stripDisallowedAttributes(clone);
      setStatus("inlining images…");
      await inlineImages(clone);
      const title = config.getTitle() || `chatExport${buildTimestamp()}`;
      const html = buildExportHtml(title, [{ role: "assistant", html: clone.innerHTML }]);
      return void downloadHtml(`${sanitizeFilename(title)}.html`, html);
    }

    const anchorElement = document.querySelector(config.scrollAnchorSelector);
    if (!anchorElement) return void alert(`${TOOL_NAME}: no conversation found on this page.`);
    const scrollContainer = findScrollableAncestor(anchorElement);

    setStatus("scanning conversation…");
    const { messageSkeletonsById, bestChunkById } = await scanConversation(scrollContainer);
    if (messageSkeletonsById.size === 0) return void alert(`${TOOL_NAME}: no messages could be captured.`);

    setStatus("cleaning up…");
    const cleanMessages = [];
    for (const [, messageSkeleton] of messageSkeletonsById) {
      cleanMessages.push({ role: messageSkeleton.role, node: buildCleanMessageNode(messageSkeleton, bestChunkById) });
    }

    setStatus("inlining images…");
    for (const message of cleanMessages) await inlineImages(message.node);

    const title = config.getTitle() || `chatExport${buildTimestamp()}`;
    const html = buildExportHtml(
      title,
      cleanMessages.map(({ role, node }) => ({ role, html: node.innerHTML }))
    );
    downloadHtml(`${sanitizeFilename(title)}_ChatExport.html`, html);
    setStatus("done ✓");
    setTimeout(() => statusToast.remove(), 1500);
  } catch (error) {
    alert(`${TOOL_NAME} failed: ${error.message}`);
  } finally {
    setTimeout(() => statusToast.remove(), 1500);
  }
})();
Self-contained — no external requests except fetching images to inline them, and only on the page you run it on.
